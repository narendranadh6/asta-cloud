GitHub Actions in 60 seconds

name: ASTA Security
on: [pull_request]
permissions:
  contents: read
  pull-requests: write
jobs:
  asta:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: narendranadh6/asta-cloud/action@v0.10.0
        with:
          path: .
          config: asta.yaml
          baseline: asta-baseline.json
          post-comment: true
